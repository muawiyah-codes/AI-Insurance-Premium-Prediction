1. cURL Terminal Request 

curl -X POST "http://localhost:8000/predict" \
  -H "Content-Type: application/json" \
  -d '{"age": 30, "weight": 72.0, "height": 1.75, "income_lpa": 12.0, "smoker": false, "city": "Mumbai", "occupation": "salaried"}'

2. Python (requests) 

import requests

payload = {
    "age": 30,
    "weight": 72.0,
    "height": 1.75,
    "income_lpa": 12.0,
    "smoker": False,
    "city": "Mumbai",
    "occupation": "salaried"
}

response = requests.post("http://localhost:8000/predict", json=payload)
data = response.json()
print("Predicted Risk Tier:", data["predicted_category"])
print("Risk Score:", data["risk_score"])
print("Annual Quote:", data["pricing"]["annual_inr"])

3. JavaScript (fetch) 

const response = await fetch("http://localhost:8000/predict", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({
    age: 30,
    weight: 72.0,
    height: 1.75,
    income_lpa: 12.0,
    smoker: false,
    city: "Mumbai",
    occupation: "salaried"
  })
});
const result = await response.json();
console.log("Risk Tier:", result.predicted_category);
